#### §4-3-3. 의사코드

```js
function collectCaseData() {
  // 1. 공통 본문 수집
  const baseYear = parseInt(document.getElementById('base-year').value, 10);
  const householdMembers = parseInt(document.getElementById('household-members').value, 10);
  const basicDeductionUsedRadio = document.querySelector('input[name="basic-deduction-used"]:checked');
  const basicDeductionUsed = basicDeductionUsedRadio?.value === 'yes';
  
  // 2. 보유 주택 수
  const householdHouseCount = document.getElementById('btn-3').classList.contains('active') ? 3 : 2;
  
  // 3. 주택별 본문 수집 (A·B·C — 보유 주택 수 만큼)
  const houseIds = householdHouseCount === 3 ? ['A', 'B', 'C'] : ['A', 'B'];
  const houses = houseIds.map(id => {
    const prefix = id.toLowerCase();  // 'a' | 'b' | 'c'
    return {
      id,
      acquisitionDate: document.getElementById(`${prefix}-acq-date`).value,
      acquisitionPrice: parseInt(document.getElementById(`${prefix}-acq-price`).value, 10),
      necessaryExpense: parseInt(document.getElementById(`${prefix}-acq-cost`).value || '0', 10),
      acquisitionRegulated: document.getElementById(`${prefix}-acq-regulated`).value === 'yes',
      residenceMonths: parseInt(document.getElementById(`${prefix}-reside-months`).value, 10),
      livingNow: document.getElementById(`${prefix}-living-now`).checked,
      expectedSaleDate: document.getElementById(`${prefix}-sale-date`).value,
      expectedSalePrice: parseInt(document.getElementById(`${prefix}-sale-price`).value, 10),
      saleRegulated: document.getElementById(`${prefix}-sale-regulated`).value === 'yes'
    };
  });
  
  // 4. salePlan 본문 수집
  const targetSaleCountRadio = document.querySelector('input[name="target-sale-count"]:checked');
  const targetSaleCountValue = targetSaleCountRadio?.value;  // '1' | '2' | 'all' | 'undecided'
  let targetSaleCount;
  if (targetSaleCountValue === 'all') targetSaleCount = householdHouseCount;
  else if (targetSaleCountValue === 'undecided') targetSaleCount = null;
  else targetSaleCount = parseInt(targetSaleCountValue, 10);
  
  const fixedSaleHouseIds = Array.from(
    document.querySelectorAll('input[name="must-sell"]:checked')
  ).map(el => el.value);
  
  const excludedHouseIds = Array.from(
    document.querySelectorAll('input[name="must-keep"]:checked')
  ).map(el => el.value);
  
  const allowYearSplitting = document.querySelector('input[name="allow-year-splitting"]')?.checked || false;
  
  const targetSaleYears = Array.from(
    document.querySelectorAll('input[name="target-sale-years"]:checked')
  ).map(el => parseInt(el.value, 10));
  
  const candidateHouseIds = houseIds.filter(id => !excludedHouseIds.includes(id));
  
  const salePlan = {
    targetSaleCount,
    candidateHouseIds,
    fixedSaleHouseIds,
    excludedHouseIds,
    allowSystemToChooseSaleTargets: fixedSaleHouseIds.length === 0,
    saleYearConstraint: allowYearSplitting ? "FLEXIBLE" : "FIXED",
    saleDeadlineYear: baseYear,
    preferredSaleYearFrom: baseYear,
    allowYearSplitting,
    targetSaleYears: targetSaleYears.length > 0 ? targetSaleYears : [baseYear]
  };
  
  // 5. caseData 객체 반환
  return {
    baseYear,
    householdMembers,
    basicDeductionUsed,
    householdHouseCount,
    isOneTimeTwoHouses: false,  // v0.3-C 본 작업지시서 미처리
    houses,
    salePlan
  };
}
```

```js
function renderScenarioTable(scenarioResults) {
  const tbody = document.getElementById('scenario-table-body');
  if (!tbody) {
    console.error('renderScenarioTable: scenario-table-body element not found');
    return;
  }
  
  // 1. rank 1 (best) → rank 6 순 정렬
  const sorted = [...scenarioResults].sort((a, b) => a.rank - b.rank);
  
  // 2. 각 시나리오별 행 생성
  const rows = sorted.map(sc => {
    const orderBadges = sc.saleOrder.map(houseId => 
      `<span class="house-dot ${houseId.toLowerCase()}">${houseId}</span>`
    ).join('<span class="order-arrow">→</span>');
    
    const perTransferTaxCells = sc.perTransferResults.map(r => 
      `<td class="tax-value">${formatKoreanCurrency(r.totalTax)}</td>`
    ).join('');
    
    const isBestClass = sc.isBest ? 'best-row' : '';
    const bestTag = sc.isBest ? '<span class="best-tag">최적</span>' : '';
    const netClass = sc.isBest ? 'net-value best' : 'net-value';
    
    return `
      <tr class="${isBestClass}">
        <td>
          <span class="order-badge">${orderBadges}</span>
          ${bestTag}
        </td>
        ${perTransferTaxCells}
        <td class="tax-value"><strong>${formatKoreanCurrency(sc.sumTotalTax)}</strong></td>
        <td class="${netClass}">${formatKoreanCurrency(sc.sumNetAfterTaxSaleAmount)}</td>
      </tr>
    `;
  }).join('');
  
  tbody.innerHTML = rows;
}

// 한국식 통화 표시 (12억 3,400만원)
function formatKoreanCurrency(value) {
  if (value === null || value === undefined) return '-';
  const eok = Math.floor(value / 100_000_000);
  const man = Math.floor((value % 100_000_000) / 10_000);
  if (eok === 0 && man === 0) return '0원';
  if (eok === 0) return `${man.toLocaleString()}만원`;
  if (man === 0) return `${eok}억원`;
  return `${eok}억 ${man.toLocaleString()}만원`;
}
```

> **`formatKoreanCurrency` 본질**: 사용자 시연 본문에서 "960,212,000원" 같은 원 단위 표기보다 "9억 6,021만원" 같은 한국식 표기가 직관 정합. 발표 시연 본문 강화.

```js
function generateSaleOrderScenarios(houses) {
  if (!Array.isArray(houses) || houses.length !== 3) {
    throw new Error(
      `generateSaleOrderScenarios: houses.length must be 3 (got ${houses?.length})`
    );
  }
  
  const ids = houses.map(h => h.id);  // ["A", "B", "C"]
  
  // 3! = 6 순열 생성
  const permutations = [];
  for (let i = 0; i < 3; i++) {
    for (let j = 0; j < 3; j++) {
      if (i === j) continue;
      for (let k = 0; k < 3; k++) {
        if (k === i || k === j) continue;
        permutations.push([ids[i], ids[j], ids[k]]);
      }
    }
  }
  
  return permutations;
  // 결과: [[A,B,C], [A,C,B], [B,A,C], [B,C,A], [C,A,B], [C,B,A]]
}
```

> **본 함수 한정 본문**: TC-S06 (3주택 모두 양도) 단일 케이스 한정. 2주택 또는 1채만 양도 분기는 본 작업지시서 미처리 — post-MVP B-034.

> **본 영속화 본문 사유**: 동일 과세연도 양도 시뮬레이션 정합 + 양도 시점 단조 비감소 정합. 본 분기 본문은 명세서 §10-7-2의 양도일 본문(`expectedSaleDate`)을 본 작업지시서에서 자동 분기로 변환 — 시연 본문 단순화.

#### §6-5-3. 의사코드

```js
function simulateScenarioWithStateTransition(caseData, saleOrder) {
  // 1. 입력 검증
  if (!Array.isArray(saleOrder) || saleOrder.length !== 3) {
    throw new Error(`simulateScenarioWithStateTransition: saleOrder.length must be 3`);
  }
  
  // 2. 양도일 자동 분기 (본 작업지시서 (가-1) 채택)
  const SALE_DATES_2026 = ["2026-03-15", "2026-06-15", "2026-09-15"];
  
  // 3. 상태전이 시뮬레이션
  const perTransferResults = [];
  let currentHouseCount = caseData.householdHouseCount;  // 초기 3
  let basicDeductionUsed = caseData.basicDeductionUsed;  // 초기 false
  
  for (let i = 0; i < saleOrder.length; i++) {
    const houseId = saleOrder[i];
    const saleDate = SALE_DATES_2026[i];
    
    const house = caseData.houses.find(h => h.id === houseId);
    if (!house) {
      throw new Error(`simulateScenarioWithStateTransition: house ${houseId} not found`);
    }
    
    // caseData_i 복사 본문 (mutation 회피)
    const caseDataI = {
      ...caseData,
      householdHouseCount: currentHouseCount,
      basicDeductionUsed,
      isOneTimeTwoHouses: false,  // 본 작업지시서 미처리
      houses: [
        {
          ...house,
          expectedSaleDate: saleDate
        }
      ]
    };
    
    // tax_engine.calculateSingleTransfer 호출 (v0.3-B 본문 그대로)
    const result = taxEngine.calculateSingleTransfer(caseDataI, houseId);
    
    perTransferResults.push({
      houseId,
      saleYear: 2026,
      saleDate,
      ...result.steps,                              // calculatedTax + taxBase + heavyRateAddition + ...
      expectedSalePrice: house.expectedSalePrice,
      issueFlags: result.issueFlags || []
    });
    
    // 상태전이
    currentHouseCount -= 1;
    basicDeductionUsed = true;  // 동일 과세연도 → 한 번 사용 후 잔여 0
  }
  
  // 4. 동일 과세연도 다중 양도 — applyFinalReturnV3 호출 (v0.3-B 본문)
  const finalReturnResult = taxEngine.applyFinalReturnV3(perTransferResults);
  
  // 5. 양도별 finalCalculatedTax 비례 분배 (v0.3-B 본문)
  taxEngine.distributeFinalTaxByShare(perTransferResults, finalReturnResult);
  
  // 6. 양도별 localIncomeTax + totalTax + netAfterTaxSaleAmount 재산출 (호출 측 책임)
  let sumTotalTax = 0;
  let sumNetAfterTaxSaleAmount = 0;
  
  for (const r of perTransferResults) {
    r.localIncomeTax = Math.floor(r.finalCalculatedTax * 0.10);
    r.totalTax = r.finalCalculatedTax + r.localIncomeTax;
    r.netAfterTaxSaleAmount = r.expectedSalePrice - r.totalTax;
    sumTotalTax += r.totalTax;
    sumNetAfterTaxSaleAmount += r.netAfterTaxSaleAmount;
  }
  
  // 7. issueFlag 통합 + v0.3-B 신규 2종 추가
  const issueFlags = perTransferResults.flatMap(r => r.issueFlags || []);
  if (finalReturnResult.method === "CLAUSE_1_AGGREGATE_PROGRESSIVE") {
    issueFlags.push("FINAL_RETURN_AGGREGATE_PROGRESSIVE_APPLIED");
  } else if (finalReturnResult.method === "CLAUSE_2_PER_TRANSFER_WITH_DAN_SEO") {
    issueFlags.push("FINAL_RETURN_DAN_SEO_APPLIED");
  }
  
  // 8. Scenario 객체 반환
  return {
    scenarioId: null,        // §6-6에서 채움
    saleOrder,
    perTransferResults,
    sumTotalTax,
    sumNetAfterTaxSaleAmount,
    rank: null,              // §6-7에서 채움
    isBest: false,           // §6-7에서 채움
    finalReturnMethod: finalReturnResult.method,
    finalReturnDiff: finalReturnResult.diff,
    issueFlags: [...new Set(issueFlags)]  // 중복 제거
  };
}
```

> **`r.steps` 본문 본질 영역**: `tax_engine.js` v0.3-B의 `calculateSingleTransfer` 반환 객체는 `result.steps` 31종(v0.3-A 27종 + v0.3-B 4종)을 본문 영속화. 본 §6-5-3 의사코드는 `result.steps`을 spread로 옮긴 후 `finalCalculatedTax` 본문이 `distributeFinalTaxByShare` 본문 mutation으로 채움 (호출 측 책임 본문 그대로).

> **계산 순서 본문**: 양도 1번째 → 2번째 → 3번째 순차 계산 (단계 4 다주택 중과 분기에서 `householdHouseCount` 본문 참조). `applyFinalReturnV3`은 3건 양도 모두 끝난 후 1회 호출 (1·2호 산출 + MAX 채택).

#### §6-6-2. 의사코드

```js
function runScenarios(caseData) {
  // 1. 본 작업지시서 (가-1) 본문 한정 검증
  if (caseData.householdHouseCount !== 3) {
    throw new Error(
      `runScenarios: only 3-house full sale supported in v0.3-C (got ${caseData.householdHouseCount})`
    );
  }
  if (caseData.salePlan.targetSaleCount !== 3) {
    throw new Error(
      `runScenarios: only targetSaleCount=3 supported in v0.3-C (got ${caseData.salePlan.targetSaleCount})`
    );
  }
  if (caseData.salePlan.allowYearSplitting) {
    throw new Error(
      `runScenarios: allowYearSplitting not supported in v0.3-C — B-034 inheritance`
    );
  }
  
  // 2. 6 양도 순서 생성
  const saleOrders = generateSaleOrderScenarios(caseData.houses);
  
  // 3. 각 시나리오 시뮬레이션
  const scenarios = saleOrders.map((saleOrder, idx) => {
    const sc = simulateScenarioWithStateTransition(caseData, saleOrder);
    sc.scenarioId = `SC-${idx + 1}`;  // SC-1 ~ SC-6
    return sc;
  });
  
  // 4. rank 산출
  return rankScenarios(scenarios);
}
```

#### §6-7-2. 의사코드

```js
function rankScenarios(scenarios) {
  // 1. sumNetAfterTaxSaleAmount 내림차순 정렬 (TaxOpt 본질 — 세후 매각금액 최대)
  const sorted = [...scenarios].sort((a, b) => 
    b.sumNetAfterTaxSaleAmount - a.sumNetAfterTaxSaleAmount
  );
  
  // 2. rank 채움 (1 ~ N)
  return sorted.map((sc, idx) => ({
    ...sc,
    rank: idx + 1,
    isBest: idx === 0
  }));
}
```

> **정렬 본질 영역 — sumNetAfterTaxSaleAmount 내림차순**: TaxOpt 본질 본문은 "세후 매각금액 최대 시나리오 추천". 의사결정 #10 D안 (TYPE_2_ORDER 시나리오 타입) `metricKey = "netAfterTaxSaleAmount"` 정합. sumTotalTax 오름차순도 동일 결과 (양도가액 동일 시).
